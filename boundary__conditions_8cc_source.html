17template <int dim, int spacedim>
18class FilteredDataOut : public DataOut<dim, spacedim>
19{
20public:
21 FilteredDataOut(const unsigned int subdomain_id)
22 : subdomain_id(subdomain_id)
23 {}
24
25 virtual typename DataOut<dim, spacedim>::cell_iterator
26 first_cell()
27 {
28 typename DataOut<dim, spacedim>::active_cell_iterator cell =
29 this->dofs->begin_active();
30 while ((cell != this->dofs->end()) &&
31 (cell->subdomain_id() != subdomain_id))
32 {
33 ++cell;
34 }
35 return cell;
36 }
37
38 virtual typename DataOut<dim, spacedim>::cell_iterator
39 next_cell(const typename DataOut<dim, spacedim>::cell_iterator &old_cell)
40 {
41 if (old_cell != this->dofs->end())
42 {
43 const IteratorFilters::SubdomainEqualTo predicate(subdomain_id);
44 return ++(
45 FilteredIterator<
46 typename DataOut<dim, spacedim>::active_cell_iterator>(predicate,
47 old_cell));
48 }
49 else
50 {
51 return old_cell;
52 }
53 }
54
55private:
56 const unsigned int subdomain_id;
57};

81{

144template <int dim>
145void
146BoundaryConditions<dim>::parse_parameters(ParameterHandler &prm)
147{
148 output_file_name = prm.get("Output file name");
149
150 std::vector<std::string> bcond_names = {"Potential",
151 "Wind function",
152 "Robin coefficients"};
153
154 for (unsigned int comp = 0; comp < n_components; ++comp)
155 {
156 for (unsigned int cond = 0; cond < bcond_names.size(); ++cond)
157 {
158 for (unsigned int slot = 0; slot < MAX_CONDITION_SLOTS; ++slot)
159 {
160 std::string label = bcond_names[cond] + " " +
161 Utilities::int_to_string(comp) + " " +
162 Utilities::int_to_string(slot) + " " +
163 Utilities::int_to_string(dim) + "d";
164
165 auto pos = comp * MAX_CONDITION_SLOTS + slot;
166
167 prm.enter_subsection(label);
168 switch (cond)
169 {
170 case BoundaryConditionType::dirichlet:
171 potentials[pos].reset(
172 new Functions::ParsedFunction<dim>(1));
173 potentials[pos]->parse_parameters(prm);
174 break;
175 case BoundaryConditionType::neumann:
176 winds[pos].reset(new Functions::ParsedFunction<dim>(dim));
177 winds[pos]->parse_parameters(prm);
178 break;
179 case BoundaryConditionType::robin:
180 robin_coeffs[pos].reset(
181 new Functions::ParsedFunction<dim>(dim));
182 robin_coeffs[pos]->parse_parameters(prm);
183 break;
184 case BoundaryConditionType::invalid:
185 default:
186 break;
187 }
188 prm.leave_subsection();
189 }
190 }
191 }

194template <int dim>
195void
196BoundaryConditions<dim>::solve_problem(bool reset_matrix)
197{
198 for (unsigned int i = 0; i < MAX_CONDITION_SLOTS; ++i)
199 {
200 get_potential(current_component, i).set_time(0);
201 get_wind(current_component, i).set_time(0);
202 get_robin_coeffs(current_component, i).set_time(0);
203 }
204
205 const types::global_dof_index n_dofs = bem.dh.n_dofs();
206 std::vector<types::subdomain_id> dofs_domain_association(n_dofs);
207 DoFTools::get_subdomain_association(bem.dh, dofs_domain_association);
208 this_cpu_set.clear();
209 this_cpu_set = bem.this_cpu_set;
210 this_cpu_set.compress();
211
212 get_phi().reinit(this_cpu_set, mpi_communicator);
213 get_dphi_dn().reinit(this_cpu_set, mpi_communicator);
214 tmp_rhs.reinit(this_cpu_set, mpi_communicator);
215 bem.robin_matrix_diagonal.reinit(this_cpu_set, mpi_communicator);
216 bem.robin_rhs.reinit(this_cpu_set, mpi_communicator);
217
218 pcout << "Computing normal vector" << std::endl;
219 if (reset_matrix)
220 {
221 bem.compute_normals();
222 }
223 prepare_bem_vectors(tmp_rhs);
224 prepare_robin_datastructs(bem.robin_matrix_diagonal, bem.robin_rhs);
225
226 bem.solve(get_phi(), get_dphi_dn(), tmp_rhs, reset_matrix);
227
228 if (!bem.can_determine_phi)
229 {
230 pcout << "Computing phi shift" << std::endl;
231 std::vector<Point<dim>> support_points(n_dofs);
232 DoFTools::map_dofs_to_support_points<dim - 1, dim>(*bem.mapping,
233 bem.dh,
234 support_points);
235 double shift = 0.0;
236 if (this_mpi_process == 0)
237 {
238 shift = get_potential(current_component, 0)
239 .value(support_points[*bem.this_cpu_set.begin()]) -
240 get_phi()(*bem.this_cpu_set.begin());
241 }
242 MPI_Bcast(&shift, 1, MPI_DOUBLE, 0, mpi_communicator);
243 vector_shift(get_phi(), shift);
244
245 pcout << "Phi shift of " << shift << std::endl;
246 }

249template <int dim>
250void
251BoundaryConditions<dim>::solve_complex_problem(bool reset_matrix)
252{
253 for (unsigned int i = 0; i < MAX_CONDITION_SLOTS; ++i)
254 {
255 // real parts - current component
256 get_potential(current_component, i).set_time(0);
257 get_wind(current_component, i).set_time(0);
258 get_robin_coeffs(current_component, i).set_time(0);
259 // imaginary parts - next component
260 get_potential(current_component + 1, i).set_time(0);
261 get_wind(current_component + 1, i).set_time(0);
262 get_robin_coeffs(current_component + 1, i).set_time(0);
263 }
264
265 const types::global_dof_index n_dofs = bem.dh.n_dofs();
266 std::vector<types::subdomain_id> dofs_domain_association(n_dofs);
267 DoFTools::get_subdomain_association(bem.dh, dofs_domain_association);
268 this_cpu_set.clear();
269 this_cpu_set = bem.this_cpu_set;
270 this_cpu_set.compress();
271
272 // real parts - current component
273 get_phi().reinit(this_cpu_set, mpi_communicator);
274 get_dphi_dn().reinit(this_cpu_set, mpi_communicator);
275 tmp_rhs.reinit(this_cpu_set, mpi_communicator);
276 bem.robin_matrix_diagonal.reinit(this_cpu_set, mpi_communicator);
277 bem.robin_rhs.reinit(this_cpu_set, mpi_communicator);
278 // imaginary parts - next component
279 get_phi(current_component + 1).reinit(this_cpu_set, mpi_communicator);
280 get_dphi_dn(current_component + 1).reinit(this_cpu_set, mpi_communicator);
281 TrilinosWrappers::MPI::Vector tmp_rhs_imag;
282 tmp_rhs_imag.reinit(this_cpu_set, mpi_communicator);
283 bem.robin_matrix_diagonal_imag.reinit(this_cpu_set, mpi_communicator);
284 bem.robin_rhs_imag.reinit(this_cpu_set, mpi_communicator);
285
286 if (reset_matrix)
287 {
288 pcout << "Computing normal vector" << std::endl;
289 bem.compute_normals();
290 }
291
292 // TODO: these calls waste the retrieval of the support points
293 // real parts - current component
294 prepare_bem_vectors(tmp_rhs);
295 // imaginary parts - next component
296 set_current_phi_component(current_component + 1);
297 prepare_bem_vectors(tmp_rhs_imag);
298 set_current_phi_component(current_component - 1);
299
300 prepare_robin_datastructs(bem.robin_matrix_diagonal,
301 bem.robin_matrix_diagonal_imag,
302 bem.robin_rhs,
303 bem.robin_rhs_imag);
304
305 bem.solve(get_phi(),
306 get_phi(current_component + 1),
307 get_dphi_dn(),
308 get_dphi_dn(current_component + 1),
309 tmp_rhs,
310 tmp_rhs_imag,
311 reset_matrix);
312
313 if (!bem.can_determine_phi)
314 {
315 pcout << "Computing phi shift of real part" << std::endl;
316 // TODO: it seems a bit wasteful to retrieve all n_dofs support pts
317 std::vector<Point<dim>> support_points(n_dofs);
318 DoFTools::map_dofs_to_support_points<dim - 1, dim>(*bem.mapping,
319 bem.dh,
320 support_points);
321 double shift = 0.0;
322 if (this_mpi_process == 0)
323 {
324 shift = get_potential(current_component, 0)
325 .value(support_points[*bem.this_cpu_set.begin()]) -
326 get_phi()(*bem.this_cpu_set.begin());
327 }
328 MPI_Bcast(&shift, 1, MPI_DOUBLE, 0, mpi_communicator);
329 vector_shift(get_phi(), shift);
330
331 pcout << "Phi shift of real part : " << shift << std::endl;
332 pcout << "Computing phi shift of imaginary part" << std::endl;
333 shift = 0.0;
334 if (this_mpi_process == 0)
335 {
336 shift = get_potential(current_component + 1, 0)
337 .value(support_points[*bem.this_cpu_set.begin()]) -
338 get_phi(current_component + 1)(*bem.this_cpu_set.begin());
339 }
340 MPI_Bcast(&shift, 1, MPI_DOUBLE, 0, mpi_communicator);
341 vector_shift(get_phi(current_component + 1), shift);
342
343 pcout << "Phi shift of imaginary part : " << shift << std::endl;
344 }

347template <int dim>
348void
349BoundaryConditions<dim>::prepare_bem_vectors(TrilinosWrappers::MPI::Vector &rhs)
350{
351 Teuchos::TimeMonitor LocalTimer(*PrepareTime);
352
353 rhs = 0;
354 get_phi() = 0;
355 get_dphi_dn() = 0;
356
357 const types::global_dof_index n_dofs = bem.dh.n_dofs();
358 std::vector<Point<dim>> support_points(n_dofs);
359 DoFTools::map_dofs_to_support_points<dim - 1, dim>(*bem.mapping,
360 bem.dh,
361 support_points);
362
363 std::vector<Point<dim>> vec_support_points(bem.gradient_dh.n_dofs());
364 DoFTools::map_dofs_to_support_points<dim - 1, dim>(*bem.mapping,
365 bem.gradient_dh,
366 vec_support_points);
367
368
369 const unsigned int dofs_per_cell = bem.fe->dofs_per_cell;
370 std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);
371 FEValues<dim - 1, dim> fe_v(*bem.mapping,
372 *bem.fe,
373 *bem.quadrature,
374 update_values | update_normal_vectors |
375 update_quadrature_points | update_JxW_values);
376
377 // pcout << "local_dof_indices elements: " << local_dof_indices.size()
378 // << std::endl;
379 // pcout << "phi elements: " << get_phi().size() << std::endl;
380 // pcout << "dphi_dn elements: " << get_phi().size() << std::endl;
381 // pcout << "support_points elements: " << support_points.size() << std::endl;
382 // pcout << "vec_support_points elements: " << vec_support_points.size()
383
384 Vector<double> coeffs(3);
385 for (const auto &cell : bem.dh.active_cell_iterators())
386 {
387 fe_v.reinit(cell);
388 cell->get_dof_indices(local_dof_indices);
389 for (unsigned int j = 0; j < bem.fe->dofs_per_cell; ++j)
390 {
391 if (this_cpu_set.is_element(local_dof_indices[j]))
392 {
393 auto slot =
394 comp_dom.manifold2bcondition_slot_map[cell->manifold_id()];
395
396 bool dirichlet = bem.dirichlet_nodes(local_dof_indices[j]) == 1;
397 if (dirichlet)
398 {
399 rhs(local_dof_indices[j]) =
400 get_potential(current_component, slot)
401 .value(support_points[local_dof_indices[j]]);
402 get_phi()(local_dof_indices[j]) = rhs(local_dof_indices[j]);
403 }
404 else
405 {
406 bool neumann = bem.neumann_nodes(local_dof_indices[j]) == 1;
407 if (neumann)
408 {
409 Vector<double> imposed_pot_grad(dim);
410 get_wind(current_component, slot)
411 .vector_value(support_points[local_dof_indices[j]],
412 imposed_pot_grad);
413
414 double tmp_dphi_dn = 0;
415 double normy = 0;
416
417 for (unsigned int d = 0; d < dim; ++d)
418 {
419 types::global_dof_index dummy =
420 bem.sub_wise_to_original[local_dof_indices[j]];
421 types::global_dof_index vec_index =
422 bem.vec_original_to_sub_wise
423 [bem.gradient_dh.n_dofs() / dim * d + dummy];
424
425 Assert(
426 bem.vector_this_cpu_set.is_element(vec_index),
427 ExcMessage(
428 "vector cpu set and cpu set are inconsistent"));
429
430 tmp_dphi_dn += imposed_pot_grad[d] *
431 bem.vector_normals_solution[vec_index];
432 normy += bem.vector_normals_solution[vec_index] *
433 bem.vector_normals_solution[vec_index];
434 }
435
436 rhs(local_dof_indices[j]) = tmp_dphi_dn;
437 get_dphi_dn()(local_dof_indices[j]) = tmp_dphi_dn;
438 }
439 else
440 {
441 rhs(local_dof_indices[j]) = 0;
442 get_phi()(local_dof_indices[j]) = 0;
443 get_dphi_dn()(local_dof_indices[j]) = 0;
444 }
445 }
446 }
447 }
448 }

451template <int dim>
452void
453BoundaryConditions<dim>::prepare_robin_datastructs(
454 TrilinosWrappers::MPI::Vector &robin_matrix_diagonal,
455 TrilinosWrappers::MPI::Vector &robin_rhs)
456{
457 Teuchos::TimeMonitor LocalTimer(*PrepareTime);
458
459 const types::global_dof_index n_dofs = bem.dh.n_dofs();
460 std::vector<Point<dim>> support_points(n_dofs);
461 DoFTools::map_dofs_to_support_points<dim - 1, dim>(*bem.mapping,
462 bem.dh,
463 support_points);
464
465 const unsigned int dofs_per_cell = bem.fe->dofs_per_cell;
466 std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);
467
468 Vector<double> coeffs(3);
469 std::set<types::global_dof_index> processed;
470 for (const auto &cell : bem.dh.active_cell_iterators())
471 {
472 if (comp_dom.manifold2bcondition_map[cell->manifold_id()] ==
473 BoundaryConditionType::robin)
474 {
475 auto slot =
476 comp_dom.manifold2bcondition_slot_map[cell->manifold_id()];
477 cell->get_dof_indices(local_dof_indices);
478 for (unsigned int j = 0; j < dofs_per_cell; ++j)
479 {
480 if (this_cpu_set.is_element(local_dof_indices[j]) &&
481 !processed.count(local_dof_indices[j]))
482 {
483 // evaluate robin coefficients
484 // coeffs(0) * phi + coeffs(1) * dphi_dn = coeffs(2)
485 get_robin_coeffs(current_component, slot)
486 .vector_value(support_points[local_dof_indices[j]], coeffs);
487 robin_matrix_diagonal(local_dof_indices[j]) =
488 coeffs(0) / coeffs(1);
489 robin_rhs(local_dof_indices[j]) = coeffs(2) / coeffs(1);
490
491 processed.insert(local_dof_indices[j]);
492 }
493 }
494 }
495 }

498template <int dim>
499void
500BoundaryConditions<dim>::prepare_robin_datastructs(
501 TrilinosWrappers::MPI::Vector &robin_matrix_diagonal,
502 TrilinosWrappers::MPI::Vector &robin_matrix_diagonal_imag,
503 TrilinosWrappers::MPI::Vector &robin_rhs,
504 TrilinosWrappers::MPI::Vector &robin_rhs_imag)
505{
506 Teuchos::TimeMonitor LocalTimer(*PrepareTime);
507
508 const types::global_dof_index n_dofs = bem.dh.n_dofs();
509 std::vector<Point<dim>> support_points(n_dofs);
510 DoFTools::map_dofs_to_support_points<dim - 1, dim>(*bem.mapping,
511 bem.dh,
512 support_points);
513
514 const unsigned int dofs_per_cell = bem.fe->dofs_per_cell;
515 std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);
516
517 Vector<double> coeffs(3);
518 Vector<double> coeffs_imag(3);
519 std::set<types::global_dof_index> processed;
520 for (const auto &cell : bem.dh.active_cell_iterators())
521 {
522 if (comp_dom.manifold2bcondition_map[cell->manifold_id()] ==
523 BoundaryConditionType::robin)
524 {
525 auto slot =
526 comp_dom.manifold2bcondition_slot_map[cell->manifold_id()];
527 cell->get_dof_indices(local_dof_indices);
528 for (unsigned int j = 0; j < dofs_per_cell; ++j)
529 {
530 if (this_cpu_set.is_element(local_dof_indices[j]) &&
531 !processed.count(local_dof_indices[j]))
532 {
533 // evaluate robin coefficients
534 // coeffs(0) * phi + coeffs(1) * dphi_dn = coeffs(2)
535 get_robin_coeffs(current_component, slot)
536 .vector_value(support_points[local_dof_indices[j]], coeffs);
537 get_robin_coeffs(current_component + 1, slot)
538 .vector_value(support_points[local_dof_indices[j]],
539 coeffs_imag);
540 std::complex<double> c0(coeffs(0), coeffs_imag(0));
541 std::complex<double> c1(coeffs(1), coeffs_imag(1));
542 std::complex<double> c2(coeffs(2), coeffs_imag(2));
543
544 std::complex<double> diag = c0 / c1;
545 std::complex<double> rhs = c2 / c1;
546
547 robin_matrix_diagonal(local_dof_indices[j]) = std::real(diag);
548 robin_matrix_diagonal_imag(local_dof_indices[j]) =
549 std::imag(diag);
550 robin_rhs(local_dof_indices[j]) = std::real(rhs);
551 robin_rhs_imag(local_dof_indices[j]) = std::imag(rhs);
552
553 processed.insert(local_dof_indices[j]);
554 }
555 }
556 }
557 }

560template <int dim>
561void
562BoundaryConditions<dim>::compute_errors(bool complex, bool current_is_real)
563{
564 Teuchos::TimeMonitor LocalTimer(*ErrorsTime);
565
566 AssertThrow(
567 complex || current_is_real,
568 ExcMessage(
569 "Called compute errors for a real problem, but specified that the current component is imaginary"));
570
571 // We still need to communicate our results to compute the errors.
572 bem.compute_gradients(get_phi(), get_dphi_dn());
573 Vector<double> localized_phi(get_phi());
574 Vector<double> localized_dphi_dn(get_dphi_dn());
575 Vector<double> localized_phi_other;
576 Vector<double> localized_dphi_dn_other;
577 if (complex)
578 {
579 if (current_is_real)
580 {
581 localized_phi_other = get_phi(current_component + 1);
582 localized_dphi_dn_other = get_dphi_dn(current_component + 1);
583 }
584 else
585 {
586 localized_phi_other = get_phi(current_component - 1);
587 localized_dphi_dn_other = get_dphi_dn(current_component - 1);
588 }
589 }
590 Vector<double> localized_gradient_solution(
591 bem.get_vector_gradients_solution());
592 Vector<double> localized_surface_gradient_solution(
593 bem.get_vector_surface_gradients_solution());
594 Vector<double> localised_normals(bem.vector_normals_solution);
595
596 Vector<double> localized_dirichlet_flags(bem.dirichlet_flags);
597 Vector<double> localized_neumann_flags(bem.neumann_flags);
598
599 // We let only the first processor do the error computations
600 if (this_mpi_process == 0)
601 {
602 pcout << "computing errors on P0" << std::endl;
603
604 Vector<double> phi_diff_cell(comp_dom.tria.n_active_cells());
605 Vector<double> phi_diff_node(bem.dh.n_dofs());
606
607 Vector<double> dphi_dn_diff_cell(comp_dom.tria.n_active_cells());
608 Vector<double> dphi_dn_diff_node(bem.dh.n_dofs());
609
610 Vector<double> gradphi_diff_cell(comp_dom.tria.n_active_cells());
611 Vector<double> gradphi_diff_node(bem.gradient_dh.n_dofs());
612
613 Vector<double> robin_diff_cell(comp_dom.tria.n_active_cells());
614 Vector<double> robin_diff_node(bem.dh.n_dofs());
615 Vector<double> coeffs(3);
616 Vector<double> coeffs_other(3);
617
618 std::vector<double> phi_refval_node(bem.dh.n_dofs());
619 std::vector<Vector<double>> gradphi_refval_node(bem.dh.n_dofs(),
620 Vector<double>(dim));
621
622 std::vector<Point<dim>> support_points(bem.dh.n_dofs());
623 DoFTools::map_dofs_to_support_points<dim - 1, dim>(*bem.mapping,
624 bem.dh,
625 support_points);
626
627 std::vector<types::global_dof_index> dofs(bem.fe->dofs_per_cell);
628 std::vector<types::global_dof_index> dofs_gradient(
629 bem.gradient_fe->dofs_per_cell);
630
631 // map material (an indirection step to boundary) to the actual expression
632 std::map<types::material_id, const Function<dim, double> *>
633 bcond_functions;
634
635 // first, build the map for the potential itself
636 for (const auto &pair : comp_dom.manifold2bcondition_map)
637 {
638 if (pair.second == BoundaryConditionType::dirichlet)
639 {

641 bcond_functions[pair.first] =
642 &get_potential(current_component, slot);
643 }
644 }
645
646 integrate_difference_based_on_material_id(*bem.mapping,
647 bem.dh,
648 localized_phi,
649 bcond_functions,
650 phi_diff_cell,
651 QGauss<(dim - 1)>(
652 2 * (2 * bem.fe->degree + 1)),
653 VectorTools::L2_norm);
654
655 bcond_functions.clear();
656 // now, build the map for the gradient
657 for (const auto &pair : comp_dom.manifold2bcondition_map)
658 {
659 if (pair.second == BoundaryConditionType::neumann)
660 {
661 auto slot = comp_dom.manifold2bcondition_slot_map[pair.first];
662 bcond_functions[pair.first] = &get_wind(current_component, slot);
663 }
664 }
665
666 integrate_difference_based_on_material_id(*bem.mapping,
667 bem.gradient_dh,
668 localized_gradient_solution,
669 bcond_functions,
670 gradphi_diff_cell,
671 QGauss<(dim - 1)>(
672 2 * (2 * bem.fe->degree + 1)),
673 VectorTools::L2_norm);
674
675 // collect the error on each dof evaluating the difference between the
676 // actual solution and that imposed by the b.condition
677 // consider that the same dof might be confronted with multiple functions;
678 // sum their errors
679
680 // process dirichlet nodes, store |phi - imposed_phi|
681 std::vector<bool> covered(bem.dh.n_dofs());
682 for (unsigned int slot = 0; slot < MAX_CONDITION_SLOTS; ++slot)
683 {
684 get_potential(current_component, slot)
685 .value_list(support_points, phi_refval_node);
686
687 std::fill(covered.begin(), covered.end(), false);
688 for (const auto &cell : bem.dh.active_cell_iterators())
689 {
690 if (cell->is_locally_owned() &&
691 comp_dom.manifold2bcondition_map[cell->manifold_id()] ==
692 BoundaryConditionType::dirichlet &&
693 comp_dom.manifold2bcondition_slot_map[cell->manifold_id()] ==
694 slot)
695 {
696 cell->get_dof_indices(dofs);
697 for (auto dof : dofs)
698 {
699 if (!covered[dof])
700 {
701 covered[dof] = true;
702
703 phi_diff_node[dof] +=
704 std::abs(phi_refval_node[dof] - localized_phi[dof]);
705 }
706 }
707 }
708 }
709 }
710
711 // process neumann nodes, store |dphi_dn - imposed_dphi_dn|
712 // also, do |grad_phi - imposed_grad_phi|?
713 for (unsigned int slot = 0; slot < MAX_CONDITION_SLOTS; ++slot)
714 {
715 get_wind(current_component, slot)
716 .vector_value_list(support_points, gradphi_refval_node);
717
718 std::fill(covered.begin(), covered.end(), false);
719 for (const auto &cell : bem.dh.active_cell_iterators())
720 {
721 if (cell->is_locally_owned() &&
722 comp_dom.manifold2bcondition_map[cell->manifold_id()] ==
723 BoundaryConditionType::neumann &&
724 comp_dom.manifold2bcondition_slot_map[cell->manifold_id()] ==
725 slot)
726 {
727 cell->get_dof_indices(dofs);
728 for (auto dof : dofs)
729 {
730 if (!covered[dof])
731 {
732 covered[dof] = true;
733
734 // compute dphi_dn from the vector_gradient_solution?
735 Tensor<1, dim> tmp_surface_gradphi, tmp_norm_gradphi,
736 tmp_gradphi, tmp_boundary_gradphi, tmp_norm;
737 double tmp_dphi_dn = 0;
738 for (unsigned int d = 0; d < dim; ++d)
739 {
740 tmp_boundary_gradphi[d] =
741 gradphi_refval_node[dof][d];
742
743 auto vec_dof = d * bem.dh.n_dofs() + dof;
744
745 tmp_norm[d] = localised_normals
746 [bem.vec_original_to_sub_wise[vec_dof]];
747
748 tmp_surface_gradphi[d] =
749 localized_surface_gradient_solution
750 [bem.vec_original_to_sub_wise[vec_dof]];
751 tmp_norm_gradphi[d] =
752 localized_dphi_dn[dof] *
753 localised_normals
754 [bem.vec_original_to_sub_wise[vec_dof]];
755 tmp_gradphi[d] =
756 tmp_surface_gradphi[d] + tmp_norm_gradphi[d];
757
758 // uses gradient_dh
759 gradphi_diff_node
760 [bem.vec_original_to_sub_wise[vec_dof]] +=
761 std::abs(
762 gradphi_refval_node
763 [bem.original_to_sub_wise[dof]][d] -
764 localized_gradient_solution
765 [bem.vec_original_to_sub_wise[vec_dof]]);
766
767 tmp_dphi_dn +=
768 localised_normals
769 [bem.vec_original_to_sub_wise[vec_dof]] *
770 gradphi_refval_node
771 [bem.original_to_sub_wise[dof]][d];
772 }
773
774 // if ((tmp_boundary_gradphi - tmp_gradphi).norm() >
775 // 1e-3)
776 // {
777 // pcout << "dof " << dof << std::endl;
778 // pcout << "\tcoords " << support_points[dof]
779 // << std::endl;
780 // pcout << "\tnormal " << tmp_norm << std::endl;
781 // pcout << "\tdphi_dn (boundary) " << tmp_dphi_dn
782 // << std::endl;
783 // pcout << "\tdphi_dn (solution) "
784 // << localized_dphi_dn[dof] << std::endl;
785 // pcout << "\tgradphi (boundary) "
786 // << gradphi_refval_node[dof] << std::endl;
787 // pcout << "\tgradphi (solution) " << tmp_gradphi
788 // << std::endl;
789 // pcout << "\tgradphi_norm " << tmp_norm_gradphi
790 // << std::endl;
791 // pcout << "\tgradphi_surface "
792 // << tmp_surface_gradphi << std::endl;
793 // }
794
795 dphi_dn_diff_node[bem.original_to_sub_wise[dof]] +=
796 std::abs(
797 tmp_dphi_dn -
798 localized_dphi_dn[bem.original_to_sub_wise[dof]]);
799 }
800 }
801 }
802 }
803 }
804
805 // process robin nodes, store |coeffs(0) * phi + coeffs(1) * dphi_dn -
806 // coeffs(2)|
807 for (unsigned int slot = 0; slot < MAX_CONDITION_SLOTS; ++slot)
808 {
809 std::fill(covered.begin(), covered.end(), false);
810 for (const auto &cell : bem.dh.active_cell_iterators())
811 {
812 if (cell->is_locally_owned() &&
813 comp_dom.manifold2bcondition_map[cell->manifold_id()] ==
814 BoundaryConditionType::robin &&
815 comp_dom.manifold2bcondition_slot_map[cell->manifold_id()] ==
816 slot)
817 {
818 cell->get_dof_indices(dofs);
819 for (auto dof : dofs)
820 {
821 if (!covered[dof])
822 {
823 get_robin_coeffs(current_component, slot)
824 .vector_value(support_points[dof], coeffs);
825 if (complex)
826 {
827 if (current_is_real)
828 {
829 get_robin_coeffs(current_component + 1, slot)
830 .vector_value(support_points[dof],
831 coeffs_other);
832 }
833 else
834 {
835 get_robin_coeffs(current_component - 1, slot)
836 .vector_value(support_points[dof],
837 coeffs_other);
838 }
839 }
840 else
841 {
842 coeffs_other = 0;
843 }
844 // assemble the complex values considering the current
845 // component's role
846 std::complex<double> c0(coeffs(0), coeffs_other(0));
847 std::complex<double> c1(coeffs(1), coeffs_other(1));
848 std::complex<double> c2(coeffs(2), coeffs_other(2));
849
850 double phi_other = 0, dphi_dn_other = 0;
851 if (complex)
852 {
853 phi_other = localized_phi_other(dof);
854 dphi_dn_other = localized_dphi_dn_other(dof);
855 }
856 std::complex<double> ph(localized_phi(dof),
857 phi_other);
858 std::complex<double> dph_dn(localized_dphi_dn(dof),
859 dphi_dn_other);
860 if (!current_is_real)
861 {
862 double tmp = c0.real();
863 c0.real(c0.imag());
864 c0.imag(tmp);
865
866 tmp = c1.real();
867 c1.real(c1.imag());
868 c1.imag(tmp);
869
870 tmp = c2.real();
871 c2.real(c2.imag());
872 c2.imag(tmp);
873
874 tmp = ph.real();
875 ph.real(ph.imag());
876 ph.imag(tmp);
877
878 tmp = dph_dn.real();
879 dph_dn.real(dph_dn.imag());
880 dph_dn.imag(tmp);
881 }
882
883 auto residual = c0 * ph + c1 * dph_dn;
884 if (current_is_real)
885 {
886 robin_diff_node[dof] +=
887 std::abs(std::real(residual - c2));
888 }
889 else
890 {
891 robin_diff_node[dof] +=
892 std::abs(std::imag(residual - c2));
893 }
894 }
895 }
896 }
897 }
898 }
899
900 VectorTools::integrate_difference(
901 *bem.mapping,
902 bem.dh,
903 dphi_dn_diff_node,
904 dealii::Functions::ZeroFunction<dim, double>(1),
905 dphi_dn_diff_cell,
906 QGauss<(dim - 1)>(2 * (2 * bem.fe->degree + 1)),
907 VectorTools::L2_norm);
908
909 VectorTools::integrate_difference(
910 *bem.mapping,
911 bem.dh,
912 robin_diff_node,
913 dealii::Functions::ZeroFunction<dim, double>(1),
914 robin_diff_cell,
915 QGauss<(dim - 1)>(2 * (2 * bem.fe->degree + 1)),
916 VectorTools::L2_norm);

920 pcout << " Number of degrees of freedom: " << bem.dh.n_dofs()
921 << std::endl;
922

924 pcout << "Phi Nodes error L_inf norm: " << phi_diff_node.linfty_norm()
925 << std::endl;
926 pcout << "Phi Cells error L_2 norm: " << phi_diff_cell.l2_norm()
927 << std::endl;
928
929 pcout << "dPhidN Nodes error L_inf norm: "
930 << dphi_dn_diff_node.linfty_norm() << std::endl;
931 pcout << "dPhidN Cells error L_2 norm: " << dphi_dn_diff_cell.l2_norm()
932 << std::endl;
933
934 pcout << "Phi Nodes Gradient error L_inf norm: "
935 << gradphi_diff_node.linfty_norm() << std::endl;
936 pcout << "Phi Cells Gradient error L_2 norm: "
937 << gradphi_diff_cell.l2_norm() << std::endl;

940 << robin_diff_node.linfty_norm() << std::endl;
941 pcout << "Robin Nodes Residual error L_2 norm: "
942 << robin_diff_cell.l2_norm() << std::endl;
943

945 "error" +
946 (current_component ? "_" + Utilities::int_to_string(current_component) :
947 "") +
948 "_vector.vtu";
949 std::vector<DataComponentInterpretation::DataComponentInterpretation>
950 data_component_interpretation(
951 dim, DataComponentInterpretation::component_is_part_of_vector);
952 DataOut<dim - 1, dim> dataout_vector;
953 dataout_vector.attach_dof_handler(bem.gradient_dh);
954 dataout_vector.add_data_vector(
955 gradphi_diff_node,

967

972 "_scalar.vtu";
973 DataOut<dim - 1, dim> dataout_scalar;
