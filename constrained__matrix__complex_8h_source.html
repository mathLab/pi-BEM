41template <class VEC, class MATRIX>
42class ConstrainedComplexOperator
43{
44public:
45 ConstrainedComplexOperator(const MATRIX &m,
46 const AffineConstraints<double> &c,
47 const AffineConstraints<double> &c_imag,
48 const IndexSet &c_cpu_set,
49 MPI_Comm comm = MPI_COMM_WORLD)
50 : constraints(c)
51 , constraints_imag(c_imag)
52 , matrix(m)
53 , constr_cpu_set(c_cpu_set)
54 , mpi_communicator(comm)
55 , n_mpi_processes(Utilities::MPI::n_mpi_processes(mpi_communicator))

63 c_dst.reinit(matrix.this_cpu_set);
64 c_dst_imag.reinit(matrix.this_cpu_set);

87template <class VEC, class MATRIX>
88void
89ConstrainedComplexOperator<VEC, MATRIX>::vmult(VEC &dst, const VEC &src) const
90{
91 auto offset = matrix.this_cpu_set.size();
92 // store localized constrained values
93 VEC loc_src(constr_cpu_set_complex);
94 loc_src = src;
95
96 // split into real, imag
97 c_src = 0;
98 c_src_imag = 0;
99 c_dst = 0;
100 c_dst_imag = 0;
101 for (auto i : matrix.this_cpu_set)
102 {
103 c_src(i) = src(i);
104 c_src_imag(i) = src(i + offset);
105 }
106 c_src.compress(VectorOperation::add);
107 c_src_imag.compress(VectorOperation::add);
108
109 matrix.vmult(c_dst, c_dst_imag, c_src, c_src_imag);
110 c_dst.compress(VectorOperation::add);
111 c_dst_imag.compress(VectorOperation::add);
112
113 for (auto i : matrix.this_cpu_set)
114 {
115 dst(i) = c_dst(i);
116 dst(i + offset) = c_dst_imag(i);
117 }
118
119 for (auto i : matrix.this_cpu_set)
120 {
121 if (constraints.is_constrained(i))
122 {
123 dst(i) += src(i) - dst(i);
124 dst(i + offset) += src(i + offset) - dst(i + offset);
125
126 for (const auto &entry : *constraints.get_constraint_entries(i))
127 {
128 dst(i) -= entry.second * loc_src(entry.first);
129 }
130
131 for (const auto &entry : *constraints_imag.get_constraint_entries(i))
132 {
133 dst(i + offset) -= entry.second * loc_src(entry.first + offset);
134 }
135 }
136 else
137 {
138 dst(i) += 0.;
139 dst(i + offset) += 0.;
140 }
141 }
142
143 dst.compress(VectorOperation::add);
144}

146template <class VEC, class MATRIX>
147void
148ConstrainedComplexOperator<VEC, MATRIX>::distribute_rhs(VEC &rhs) const
149{
150 for (auto i : matrix.this_cpu_set)
151 {
152 if (constraints.is_constrained(i))
153 {
154 rhs(i) = constraints.get_inhomogeneity(i);
155 rhs(i + matrix.this_cpu_set.size()) =
156 constraints_imag.get_inhomogeneity(i);
157 }
158 else
159 {
160 rhs(i) = rhs(i);
161 rhs(i + matrix.this_cpu_set.size()) =
162 rhs(i + matrix.this_cpu_set.size());
163 }
164 }
165}

250template <class VEC, class MATRIX>
251void
252BlockConstrainedComplexOperator<VEC, MATRIX>::vmult(VEC &dst,
253 const VEC &src) const
254{
255 // store localized constrained values
256 std::vector<IndexSet> parallel_partitioning{constr_cpu_set, constr_cpu_set};
257 VEC localized_src;
258 localized_src.reinit(parallel_partitioning, mpi_communicator);
259 for (unsigned int i = 0; i < src.n_blocks(); ++i)
260 {
261 localized_src.block(i) = src.block(i);
262 }
263
264 matrix.vmult(dst.block(0), dst.block(1), src.block(0), src.block(1));
265 dst.compress(VectorOperation::add);
266
267 for (auto i : matrix.this_cpu_set)
268 {
269 if (constraints.is_constrained(i))
270 {
271 dst.block(0)(i) += src.block(0)(i) - dst.block(0)(i);
272 for (const auto &entry : *constraints.get_constraint_entries(i))
273 {
274 dst.block(0)(i) -=
275 entry.second * localized_src.block(0)(entry.first);
276 }
277
278 dst.block(1)(i) += src.block(1)(i) - dst.block(1)(i);
279 for (const auto &entry : *constraints_imag.get_constraint_entries(i))
280 {
281 dst.block(1)(i) -=
282 entry.second * localized_src.block(1)(entry.first);
283 }
284 }
285 }
286
287 dst.compress(VectorOperation::add);
288}

293{
294 for (auto i : matrix.this_cpu_set)
