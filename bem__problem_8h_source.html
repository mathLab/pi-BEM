472 FullMatrix<double>
473 get_system_matrix() const
474 {
475 Vector<double> localized_dirichlet_nodes(dirichlet_nodes);
476 Vector<double> localized_neumann_nodes(neumann_nodes);
477 Vector<double> localized_robin_nodes(robin_nodes);
478 Vector<double> localized_alpha(alpha);
479 Vector<double> localized_robin_matrix_diagonal(robin_matrix_diagonal);
480
481 SparsityPattern spattern(dh.n_dofs(), dh.n_dofs(), dh.n_dofs());
482 for (unsigned int i = 0; i < dh.n_dofs(); ++i)
483 {
484 for (unsigned int j = 0; j < dh.n_dofs(); ++j)
485 {
486 spattern.add(i, j);
487 }
488 }
489 spattern.compress();
490
491 SparseMatrix<double> localized_dirichlet_matrix(spattern);
492 localized_dirichlet_matrix.copy_from(dirichlet_matrix);
493 SparseMatrix<double> localized_neumann_matrix(spattern);
494 localized_neumann_matrix.copy_from(neumann_matrix);
495
496 FullMatrix<double> ret(dh.n_dofs(), dh.n_dofs());
497
498 for (unsigned int i = 0; i < dh.n_dofs(); ++i)
499 {
500 for (unsigned int j = 0; j < dh.n_dofs(); ++j)
501 {
502 if (localized_dirichlet_nodes(j))
503 {
504 ret.set(i, j, localized_neumann_matrix(i, j));
505 if (i == j)
506 {
507 ret.add(i, j, localized_alpha(i));
508 }
509 }
510 else if (localized_neumann_nodes(j))
511 {
512 ret.set(i, j, -localized_dirichlet_matrix(i, j));
513 }
514 else
515 {
516 ret.set(i, j, localized_neumann_matrix(i, j));
517 ret.add(i,
518 j,
519 localized_dirichlet_matrix(i, j) *
520 localized_robin_matrix_diagonal(j));
521 if (i == j)
522 {
523 ret.add(i, j, localized_alpha(i));
524 }
525 }
526 }
527 }
528
529 return ret;
530 }

535 FullMatrix<double>
536 get_system_matrix_complex() const
537 {
538 Vector<double> localized_dirichlet_nodes(dirichlet_nodes);
539 Vector<double> localized_neumann_nodes(neumann_nodes);
540 Vector<double> localized_robin_nodes(robin_nodes);
541 Vector<double> localized_alpha(alpha);
542 Vector<double> localized_robin_matrix_diagonal(robin_matrix_diagonal);
543 Vector<double> localized_robin_matrix_diagonal_imag(
544 robin_matrix_diagonal_imag);
545
546 SparsityPattern spattern(dh.n_dofs(), dh.n_dofs(), dh.n_dofs());
547 for (unsigned int i = 0; i < dh.n_dofs(); ++i)
548 {
549 for (unsigned int j = 0; j < dh.n_dofs(); ++j)
550 {
551 spattern.add(i, j);
552 }
553 }
554 spattern.compress();
555
556 SparseMatrix<double> localized_dirichlet_matrix(spattern);
557 localized_dirichlet_matrix.copy_from(dirichlet_matrix);
558 SparseMatrix<double> localized_neumann_matrix(spattern);
559 localized_neumann_matrix.copy_from(neumann_matrix);
560
561 FullMatrix<double> ret(2 * dh.n_dofs(), 2 * dh.n_dofs());
562
563 for (unsigned int i = 0; i < dh.n_dofs(); ++i)
564 {
565 for (unsigned int j = 0; j < dh.n_dofs(); ++j)
566 {
567 if (localized_dirichlet_nodes(j))
568 {
569 ret.set(i, j, localized_neumann_matrix(i, j));
570 ret.set(i + dh.n_dofs(),
571 j + dh.n_dofs(),
572 localized_neumann_matrix(i, j));
573 if (i == j)
574 {
575 ret.add(i, j, localized_alpha(i));
576 ret.add(i + dh.n_dofs(),
577 j + dh.n_dofs(),
578 localized_alpha(i));
579 }
580 }
581 else if (localized_neumann_nodes(j))
582 {
583 ret.set(i, j, -localized_dirichlet_matrix(i, j));
584 ret.set(i + dh.n_dofs(),
585 j + dh.n_dofs(),
586 -localized_dirichlet_matrix(i, j));
587 }
588 else
589 {
590 ret.set(i, j, localized_neumann_matrix(i, j));
591 ret.set(i + dh.n_dofs(),
592 j + dh.n_dofs(),
593 localized_neumann_matrix(i, j));
594
595 // in the diagonal blocks, it's just the real part of the robin
596 // datastructs
597 ret.add(i,
598 j,
599 localized_dirichlet_matrix(i, j) *
600 localized_robin_matrix_diagonal(j));
601 ret.add(i + dh.n_dofs(),
602 j + dh.n_dofs(),
603 localized_dirichlet_matrix(i, j) *
604 localized_robin_matrix_diagonal(j));
605 // but the in the other quadrants, it's the imaginary part with
606 // flipping of the sign
607 ret.add(i,
608 j + dh.n_dofs(),
609 -localized_dirichlet_matrix(i, j) *
610 localized_robin_matrix_diagonal_imag(j));
611 ret.add(i + dh.n_dofs(),
612 j,
613 localized_dirichlet_matrix(i, j) *
614 localized_robin_matrix_diagonal_imag(j));
615
616 if (i == j)
617 {
618 ret.add(i, j, localized_alpha(i));
619 ret.add(i + dh.n_dofs(),
620 j + dh.n_dofs(),
621 localized_alpha(i));
622 }
623 }
624 }
625 }
626
627 return ret;
628 }
