56void
57do_integrate_difference_based_on_material_id(
58 const hp::MappingCollection<dim, spacedim> &mapping,
59 const DoFHandler<dim, spacedim> &dof,
60 const InVector &fe_function,
61 // const Function<spacedim, typename InVector::value_type> &exact_solution,
62 const std::map<types::material_id,
63 const Function<spacedim, typename InVector::value_type> *>
64 &exact_solutions,
65 OutVector &difference,
66 const dealii::hp::QCollection<dim> &q,
67 const VectorTools::NormType &norm,
68 const Function<spacedim> *weight,
69 const double exponent_1)
70{
71 using Number = typename InVector::value_type;
72 // we mark the "exponent" parameter to this function "const" since it is
73 // strictly incoming, but we need to set it to something different later
74 // on, if necessary, so have a read-write version of it:
75 double exponent = exponent_1;
76
77 const unsigned int n_components = dof.get_fe(0).n_components();
78
79#ifdef DEBUG
80 for (const auto &pair : exact_solutions)
81 {
82 Assert(pair.second->n_components == n_components,
83 ExcDimensionMismatch(pair.second->n_components, n_components));
84 }
85#endif
86
87 if (weight != nullptr)
88 {
89 Assert((weight->n_components == 1) ||
90 (weight->n_components == n_components),
91 ExcDimensionMismatch(weight->n_components, n_components));
92 }
93
94 difference.reinit(dof.get_triangulation().n_active_cells());
95
96 switch (norm)
97 {
98 case VectorTools::L2_norm:
99 case VectorTools::H1_seminorm:
100 case VectorTools::H1_norm:
101 case VectorTools::Hdiv_seminorm:
102 exponent = 2.;
103 break;
104
105 case VectorTools::L1_norm:
106 exponent = 1.;
107 break;
108
109 default:
110 break;
111 }
112
113 UpdateFlags update_flags =
114 UpdateFlags(update_quadrature_points | update_JxW_values);
115 switch (norm)
116 {
117 case VectorTools::H1_seminorm:
118 case VectorTools::Hdiv_seminorm:
119 case VectorTools::W1p_seminorm:
120 case VectorTools::W1infty_seminorm:
121 update_flags |= UpdateFlags(update_gradients);
122 if (spacedim == dim + 1)
123 update_flags |= UpdateFlags(update_normal_vectors);
124
125 break;
126
127 case VectorTools::H1_norm:
128 case VectorTools::W1p_norm:
129 case VectorTools::W1infty_norm:
130 update_flags |= UpdateFlags(update_gradients);
131 if (spacedim == dim + 1)
132 update_flags |= UpdateFlags(update_normal_vectors);
133 DEAL_II_FALLTHROUGH;
134
135 default:
136 update_flags |= UpdateFlags(update_values);
137 break;
138 }
139
140 const dealii::hp::FECollection<dim, spacedim> &fe_collection =
141 dof.get_fe_collection();
142 VectorTools::internal::IDScratchData<dim, spacedim, Number> data(
143 mapping, fe_collection, q, update_flags);
144
145 // loop over all cells
146 for (const auto &cell : dof.active_cell_iterators())
147 if (cell->is_locally_owned())
148 {
149 // initialize for this cell
150 data.x_fe_values.reinit(cell);
151
152 const dealii::FEValues<dim, spacedim> &fe_values =
153 data.x_fe_values.get_present_fe_values();
154 const unsigned int n_q_points = fe_values.n_quadrature_points;
155 data.resize_vectors(n_q_points, n_components);
156
157 if (update_flags & update_values)
158 fe_values.get_function_values(fe_function, data.function_values);
159 if (update_flags & update_gradients)
160 fe_values.get_function_gradients(fe_function, data.function_grads);
161
162 const auto found = exact_solutions.find(cell->material_id());
163 if (found != exact_solutions.end())
164 {
165 difference(cell->active_cell_index()) = VectorTools::internal::
166 integrate_difference_inner<dim, spacedim, Number>(*found->second,
167 norm,
168 weight,
169 update_flags,
170 exponent,
171 n_components,
172 data);
173 // std::cout << "using boundary-selected integration" << std::endl;
174 }
175 else
176 {
177 difference(cell->active_cell_index()) = 0;
178 // std::cout
179 // << "skipping cell difference due to missing function
180 // assignment"
181 // << std::endl;
182 }
183 }
184 else
185 // the cell is a ghost cell or is artificial. write a zero into the
186 // corresponding value of the returned vector
187 difference(cell->active_cell_index()) = 0;
188}
