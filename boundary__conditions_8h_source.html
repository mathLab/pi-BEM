298 TrilinosWrappers::MPI::BlockVector
299 get_phi_components(const std::vector<unsigned int> &component_idxs)
300 {
301 TrilinosWrappers::MPI::BlockVector ret;
302 std::vector<IndexSet> indexSets(n_components);
303 for (auto comp : component_idxs)
304 {
305 for (auto i : this_cpu_set)
306 {
307 indexSets[comp].add_index(comp * this_cpu_set.size() + i);
308 }
309 indexSets[comp].compress();
310 }
311
312 ret.reinit(indexSets, mpi_communicator);
313
314 for (auto comp : component_idxs)
315 {
316 for (auto i : this_cpu_set)
317 {
318 ret[comp * this_cpu_set.size() + i] = get_phi(comp)[i];
319 }
320 }
321
322 ret.compress(VectorOperation::add);
323
324 return ret;
325 }

331 TrilinosWrappers::MPI::BlockVector

335 for (unsigned int i = 0; i < n_components; ++i)
336 {
337 component_idxs[i] = i;
338 }
339
340 return get_phi_components(component_idxs);
