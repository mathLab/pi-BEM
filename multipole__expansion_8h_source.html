179 static void
180 spherical_coords(const dealii::Point<3> &center,
181 const dealii::Point<3> &other,
182 dealii::Point<3> &blockRelPos,
183 double &rho,
184 double &cos_alpha,
185 double &beta)
186 {
187 blockRelPos = other - center;
188 rho = blockRelPos.norm();
189 cos_alpha = blockRelPos(2) / rho;
190 beta = atan2(blockRelPos(1), blockRelPos(0));
191 }
192
193 static FullMatrix<double>
194 A_n_m_Matrix(unsigned int dim)
195 {
196 FullMatrix<double> A_n_m(dim + 1, dim + 1);
197 for (unsigned int n = 0; n < dim + 1; n++)
198 {
199 for (unsigned int m = 0; m < n + 1; m++)
200 {
201 double f1 = 1.;
202 double f2 = 1.;
203
204 for (int ii = n - m; ii > 0; ii--)
205 {
206 f1 *= ii;
207 }
208
209 for (int ii = n + m; ii > 0; ii--)
210 {
211 f2 *= (ii);
212 }
213
214 A_n_m(n, m) = pow(-1., double(n)) / sqrt(f1 * f2);
215 }
216 }
217
218 return A_n_m;
219 }
